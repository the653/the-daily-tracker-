<script>
  // Dedicated multi-device real-time sync database
  const firebaseConfig = {
    databaseURL: "https://day-tracker-live-default-rtdb.firebaseio.com"
  };
  firebase.initializeApp(firebaseConfig);
  const db = firebase.database();
  const trackerRef = db.ref('family_day_tracker');

  let defaultUsers = ['Milan', 'Madhur', 'Suresha', 'Rekha'];
  let currentActiveUser = 'Milan';
  let trackerData = {
    users: defaultUsers,
    schedules: {}
  };

  window.onload = function() {
    const today = new Date().toISOString().split('T')[0];
    document.getElementById('currentDate').value = today;

    // Real-time listener: triggers instantly whenever ANY phone updates
    trackerRef.on('value', (snapshot) => {
      const remote = snapshot.val();
      if (remote) {
        trackerData.users = remote.users || defaultUsers;
        trackerData.schedules = remote.schedules || {};
      } else {
        // Initialize cloud database if empty
        saveToCloud();
      }
      setSyncStatus('🟢 Live Synced with All Devices', '#16a34a');
      renderUserTabs();
      renderDayTasks();
    }, (error) => {
      setSyncStatus('⚠️ Cloud Offline - Check Internet', '#dc2626');
    });
  };

  function setSyncStatus(msg, color) {
    const el = document.getElementById('sync-status');
    el.innerText = msg;
    el.style.color = color;
  }

  function saveToCloud() {
    setSyncStatus('⏳ Syncing...', '#d97706');
    trackerRef.set(trackerData).then(() => {
      setSyncStatus('🟢 Live Synced with All Devices', '#16a34a');
    }).catch(() => {
      setSyncStatus('⚠️ Sync Error', '#dc2626');
    });
  }

  function renderUserTabs() {
    const tabContainer = document.getElementById('user-tabs');
    tabContainer.innerHTML = '';
    
    if (!trackerData.users.includes(currentActiveUser)) {
      currentActiveUser = trackerData.users[0] || 'Milan';
    }

    trackerData.users.forEach(user => {
      const btn = document.createElement('button');
      btn.className = `user-tab ${user === currentActiveUser ? 'active' : ''}`;
      btn.innerText = `👤 ${user}`;
      btn.onclick = () => {
        currentActiveUser = user;
        renderUserTabs();
        renderDayTasks();
      };
      tabContainer.appendChild(btn);
    });
  }

  function promptNewUser() {
    const name = prompt("Enter the name of the new Day Tracker user:");
    if (name && name.trim() !== '') {
      const trimmed = name.trim();
      if (!trackerData.users.includes(trimmed)) {
        trackerData.users.push(trimmed);
        currentActiveUser = trimmed;
        saveToCloud();
      }
    }
  }

  function getSelectedDate() {
    return document.getElementById('currentDate').value;
  }

  function getActiveUserTasks() {
    const date = getSelectedDate();
    if (!trackerData.schedules) trackerData.schedules = {};
    if (!trackerData.schedules[currentActiveUser]) {
      trackerData.schedules[currentActiveUser] = {};
    }
    if (!trackerData.schedules[currentActiveUser][date]) {
      trackerData.schedules[currentActiveUser][date] = [];
    }
    return trackerData.schedules[currentActiveUser][date];
  }

  function renderDayTasks() {
    const tasks = getActiveUserTasks();
    const tbody = document.getElementById('task-tbody');
    tbody.innerHTML = '';

    tasks.forEach((task, index) => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td class="w-sno" style="text-align:center; font-weight:600;">${index + 1}</td>
        <td><input type="text" value="${escapeHtml(task.name || '')}" oninput="updateTask(${index}, 'name', this.value)" placeholder="Enter ${currentActiveUser}'s task..."></td>
        <td class="w-priority">
          <select onchange="updateTask(${index}, 'priority', this.value)">
            <option ${task.priority === 'Medium' ? 'selected' : ''}>Medium</option>
            <option ${task.priority === 'High' ? 'selected' : ''}>High</option>
            <option ${task.priority === 'Low' ? 'selected' : ''}>Low</option>
          </select>
        </td>
        <td class="w-status">
          <select onchange="updateTask(${index}, 'status', this.value)">
            <option ${task.status === 'Not Started' ? 'selected' : ''}>Not Started</option>
            <option ${task.status === 'In Progress' ? 'selected' : ''}>In Progress</option>
            <option ${task.status === 'Deferred' ? 'selected' : ''}>Deferred</option>
            <option ${task.status === 'Completed' ? 'selected' : ''}>Completed</option>
          </select>
        </td>
        <td class="w-date"><input type="date" value="${task.deferredDate || ''}" onchange="updateTask(${index}, 'deferredDate', this.value)"></td>
        <td><input type="text" value="${escapeHtml(task.remarks || '')}" oninput="updateTask(${index}, 'remarks', this.value)" placeholder="Remarks/Notes..."></td>
        <td class="w-action"><button class="btn btn-delete" onclick="deleteTask(${index})">✕</button></td>
      `;
      tbody.appendChild(tr);
    });

    updateKPIs(tasks);
  }

  function addNewTask() {
    const tasks = getActiveUserTasks();
    tasks.push({
      name: '',
      priority: 'Medium',
      status: 'Not Started',
      deferredDate: '',
      remarks: ''
    });
    saveToCloud();
  }

  let debounceTimer = null;
  function updateTask(index, field, value) {
    const tasks = getActiveUserTasks();
    if (tasks[index]) {
      tasks[index][field] = value;
      if (field === 'status') updateKPIs(tasks);
      
      clearTimeout(debounceTimer);
      debounceTimer = setTimeout(() => {
        saveToCloud();
      }, 300);
    }
  }

  function deleteTask(index) {
    const tasks = getActiveUserTasks();
    tasks.splice(index, 1);
    saveToCloud();
  }

  function updateKPIs(tasks) {
    const total = tasks.length;
    let notStarted = 0, inProgress = 0, deferred = 0, completed = 0;

    tasks.forEach(t => {
      if (t.status === 'Not Started') notStarted++;
      else if (t.status === 'In Progress') inProgress++;
      else if (t.status === 'Deferred') deferred++;
      else if (t.status === 'Completed') completed++;
    });

    const percent = total > 0 ? Math.round((completed / total) * 100) : 0;

    document.getElementById('kpi-total').innerText = total;
    document.getElementById('kpi-notstarted').innerText = notStarted;
    document.getElementById('kpi-inprogress').innerText = inProgress;
    document.getElementById('kpi-deferred').innerText = deferred;
    document.getElementById('kpi-completed').innerText = completed;
    document.getElementById('percent-label').innerText = percent + '%';
    document.getElementById('progress-bar').style.width = percent + '%';
  }

  function escapeHtml(str) {
    return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
  }
</script>
